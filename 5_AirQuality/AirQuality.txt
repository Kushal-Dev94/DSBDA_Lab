# Drop unnecessary columns
data = data.drop(['Unnamed: 15', 'Unnamed: 16'], axis=1)

# Convert string to float where applicable
data['CO(GT)'] = data['CO(GT)'].str.replace(',', '.').astype(float)
data['C6H6(GT)'] = data['C6H6(GT)'].str.replace(',', '.').astype(float)
data['Temperature'] = data['Temperature'].str.replace(',', '.').astype(float)

data = data.drop_duplicates()

# Check missing values
data.isna().sum()

data = data.fillna(data.mean())

# Drop rows with missing values
data = data.dropna()

# Check missing values again
data.isna().sum()

# Convert 'Absolute Humidity' to percentage
data['Absolute Humidity'] = data['Absolute Humidity'].multiply(100)

# Remove outliers for selected columns
def remove_outliers(column):
    Q1 = column.quantile(0.25)
    Q3 = column.quantile(0.75)
    IQR = Q3 - Q1
    threshold = 1.5 * IQR
    outlier_mask = (column < Q1 - threshold) | (column > Q3 + threshold)
    return column[~outlier_mask]

# List of columns to clean for outliers
columns_to_clean = ['Temperature', 'Relative Humidity', 'Absolute Humidity', 'PT08.S4(NO2)', 'PT08.S5(O3)', 
                    'C6H6(GT)', 'PT08.S2(NMHC)', 'PT08.S1(CO)']

for col in columns_to_clean:
    data[col] = remove_outliers(data[col])

# Add Year and Month columns
data['Year'] = pd.to_datetime(data['Date']).dt.year
data['Month'] = pd.to_datetime(data['Date']).dt.month
