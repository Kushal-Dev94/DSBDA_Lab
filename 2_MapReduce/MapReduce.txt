#!/bin/bash

# Set Hadoop environment variables
export HADOOP_HOME=/usr/local/hadoop
export JAVA_HOME=/usr/lib/jvm/java-8-openjdk-amd64
export PATH=$PATH:$HADOOP_HOME/bin:$HADOOP_HOME/sbin

# Start Hadoop in Pseudo-distributed mode
start-dfs.sh
start-yarn.sh

# Create the 'analyzelogs' folder in home directory and move the necessary files
mkdir -p ~/analyzelogs
cd ~/analyzelogs

# Download log file (Replace with actual URL if available)
# wget <log_file_url> -O access_log_short.csv

# Convert access_log.txt to CSV using LibreCalc (manual step)
# Open access_log.txt in LibreCalc, set rows as per the format, then export as CSV.

# Set CLASSPATH for Hadoop and Java files
export CLASSPATH="$HADOOP_HOME/share/hadoop/mapreduce/hadoop-mapreduce-client-core-2.9.0.jar:$HADOOP_HOME/share/hadoop/mapreduce/hadoop-mapreduce-client-common-2.9.0.jar:$HADOOP_HOME/share/hadoop/common/hadoop-common-2.9.0.jar:~/mapreduce_yogesh/SalesCountry/*:$HADOOP_HOME/lib/*"

# Compile the Mapper, Reducer, and Driver Java files
javac -d . SalesMapper.java SalesCountryReducer.java SalesCountryDriver.java

# Create Manifest.txt
sudo gedit Manifest.txt
# Add the following line to Manifest.txt:
# Main-Class: SalesCountry.SalesCountryDriver

# Create a JAR file
jar -cfm analyzelogs.jar Manifest.txt SalesCountry/*.class

# Create HDFS directories and upload input file
hdfs dfs -mkdir ~/input2000
cp access_log_short.csv ~/input2000/
$HADOOP_HOME/bin/hdfs dfs -put ~/input2000 /input2000

# Run the MapReduce job
$HADOOP_HOME/bin/hadoop jar analyzelogs.jar /input2000 /output2000

# View the output
$HADOOP_HOME/bin/hdfs dfs -cat /output2000/part-00000

# Optionally, clean up
# $HADOOP_HOME/bin/hdfs dfs -rm -r /input2000 /output2000
